
// Permit lifecycle signals and the single bound-call channel. All other
// JavaScript messages are rejected before framework special handlers.
func ateneaSSHIngressMessage(message string) bool {
	return message == "DomReady" || message == "runtime:ready" || message == "Q" || (len(message) >= 2 && len(message) <= 4096 && message[0] == 'C')
}
